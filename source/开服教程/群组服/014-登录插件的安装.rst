我们选择authmereload方案。 

创建数据库

.. code-block:: sql 

    create database  d_authme CHARACTER SET utf8 COLLATE utf8_general_ci;


安装
==================================================

`authmereload 下载 <https://github.com/AuthMe/AuthMeReloaded/releases/download/5.6.0/AuthMe-5.6.0.jar>`_ 


.. code-block:: bash 

    # 下载
    wget https://github.com/AuthMe/AuthMeReloaded/releases/download/5.6.0/AuthMe-5.6.0.jar
    # 放到login分区的plugins目录，然后重启， 修改配置文件如下。 
    root@mc:/home/mc/instances/login/plugins/AuthMe# diff config.yml config.yml.default
    4c4
    <     backend: MARIADB
    ---
    >     backend: SQLITE
    13c13
    <     mySQLUseSSL: false
    ---
    >     mySQLUseSSL: true
    22c22
    <     mySQLUsername: mc
    ---
    >     mySQLUsername: authme
    24c24
    <     mySQLPassword: 'mc_panda_142857'
    ---
    >     mySQLPassword: '12345'
    26c26
    <     mySQLDatabase: d_authme
    ---
    >     mySQLDatabase: authme
    70c70
    <     poolSize: 100
    ---
    >     poolSize: 10
    107c107
    <         enabled: true
    ---
    >         enabled: false
    110c110
    <         timeout: 1800
    ---
    >         timeout: 10
    113c113
    <     messagesLanguage: zhcn
    ---
    >     messagesLanguage: en
    124c124
    <     serverName: Your Minecraft Server
    ---
    >     serverName: Your Minecraft Server
    304c304
    <     useWelcomeMessage: false
    ---
    >     useWelcomeMessage: true



    # 复制到其他的分区
    cp -r AuthMe* /home/mc/instances/dp1/plugins/
    cp -r AuthMe* /home/mc/instances/sc1/plugins/
    cp -r AuthMe* /home/mc/instances/sc2/plugins/

    # 重启几个分区
    systemctl restart mc_dp1 mc_sc1 mc_sc2
